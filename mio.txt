-- FIX: Measure Name prefixed with REPT(UNICHAR(8203), Sort)
--      Makes each Measure Name globally unique for PBI Service.
--      UNICHAR(8203) = Zero Width Space, invisible in matrix columns.
--      Sort is already unique per row so prefix is unique per row.
-- Is Visible logic: CUMULATIVE forecast visibility
--   _MaxP <= 3  : no forecasts shown
--   _MaxP <= 6  : P3F visible
--   _MaxP <= 9  : P3F + P6F visible
--   _MaxP >  9  : P3F + P6F + P9F all visible
-- ============================================================


VAR PLABEL = VALUE(SELECTEDVALUE('Dim Period'[Period]))

-- Check if each forecast file has data loaded
VAR _P3FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P3F"
   ) > 0


VAR _P6FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P6F"
   ) > 0


VAR _P9FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P9F"
   ) > 0


VAR _MaxP =
    CALCULATE(
        MAX('FS SPL (fact)'[Period]),
        ALL('FS SPL (fact)'),
        'FS SPL (fact)'[TBL] = "BEAM"
    )

-- BOTH conditions must be true:
-- 1. MaxP has reached the threshold
-- 2. The forecast file actually exists


VAR _VisibleForecasts =
    SWITCH(TRUE(),
    
        _MaxP <= 3  , "",
        _MaxP <= 6 , "P3F",
       _MaxP <= 9 , "P3F|P6F",
        "P3F|P6F|P9F"
    )

VAR _WithVisibility =
    ADDCOLUMNS(
        _Rows,
        "Is Visible",
        IF(
            [Forecast Flag] = "ALL"
            || CONTAINSSTRING(_VisibleForecasts, [Forecast Flag]),
            1,
            0
        )
    )

-- Apply UNICHAR(8203) prefix keyed to Sort to make Measure Name globally unique
-- Invisible in matrix column headers. Satisfies PBI Service one-to-one sort rule.
RETURN
    SELECTCOLUMNS(
        _WithVisibility,
        "Measure Name",  REPT(UNICHAR(8203), [Sort]) & [Measure Name],
        "Measure Ref",   [Measure Ref],
        "Sort",          [Sort],
        "View ID",       [View ID],
        "Group",         [Group],
        "Forecast Flag", [Forecast Flag],
        "Is Visible",    [Is Visible]
    )

View Selector = -- ============================================================
-- VIEW SELECTOR TABLE
-- Purpose: Drives the view slicer on the matrix page
-- User picks one view (single select) which controls
-- which columns appear in the matrix via Measure Selector
--
-- Views:
--   1 = Basic View      - single period actuals + budget + forecast
--   2 = Progressive View - YTD only, same sequence as Basic
--   3 = All View        - full set single period then YTD
--
-- HOW TO USE:
--   1. Add this table as a slicer on the matrix page
--   2. Use View Name column as the slicer field
--   3. Set slicer to single select
--   4. The Measure Selector table filters by View ID
-- ============================================================
DATATABLE(
    "View ID",   INTEGER,
    "View Name", STRING,
    "View Sort", INTEGER,
    {
        {1, "Current Month",1},
        {2, "Progressive", 2},
        {3, "Combined",    3},
         {4, "All",        4}
    }
)
