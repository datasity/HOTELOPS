-- MEASURE SELECTOR TABLE
-- Measure Name prefixed with REPT(UNICHAR(8203), Sort)
--      Makes each Measure Name globally unique for PBI Service.
--      UNICHAR(8203) = Zero Width Space, invisible in matrix columns.
--      Sort must be unique per row so the prefix is unique per row.
-- Is Visible logic: CUMULATIVE forecast visibility
--   _MaxP <= 3  : no forecasts shown
--   _MaxP <= 6  : P3F visible
--   _MaxP <= 9  : P3F + P6F visible
--   _MaxP >  9  : P3F + P6F + P9F all visible
-- VIEW IDs:
--   1 = Current Month, 2 = Progressive, 3 = Combined, 4 = All
--   5 = P3F View, 6 = P6F View, 7 = P9F View
-- NEW VIEWS USE SORT 910+ (never reuse a Sort number)
-- ============================================================

VAR PLABEL = VALUE(SELECTEDVALUE('Dim Period'[Period]))

VAR _P3FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P3F"
   ) > 0

VAR _P6FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P6F"
   ) > 0

VAR _P9FExists =
   CALCULATE(
       COUNTROWS('FS SPL (fact)'),
       ALL('FS SPL (fact)'),
       'FS SPL (fact)'[Year Flag] = "P9F"
   ) > 0

VAR _MaxP =
    CALCULATE(
        MAX('FS SPL (fact)'[Period]),
        ALL('FS SPL (fact)'),
        'FS SPL (fact)'[TBL] = "BEAM"
    )

VAR _VisibleForecasts =
    SWITCH(TRUE(),
        _MaxP <= 3, "",
        _MaxP <= 6, "P3F",
        _MaxP <= 9, "P3F|P6F",
        "P3F|P6F|P9F"
    )

VAR _WithVisibility =
    ADDCOLUMNS(
        _Rows,
        "Is Visible",
        IF(
            [Forecast Flag] = "ALL"
            || CONTAINSSTRING(_VisibleForecasts, [Forecast Flag]),
            1,
            0
        )
    )

RETURN
    SELECTCOLUMNS(
        _WithVisibility,
        "Measure Name",  REPT(UNICHAR(8203), [Sort]) & [Measure Name],
        "Measure Ref",   [Measure Ref],
        "Sort",          [Sort],
        "View ID",       [View ID],
        "Group",         [Group],
        "Forecast Flag", [Forecast Flag],
        "Is Visible",    [Is Visible]
    )
