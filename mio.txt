-- ============================================================
-- VIEW SELECTOR TABLE
-- Drives the view slicer on the matrix page (single select).
-- The Measure Selector filters by View ID.
--   1 = Current Month
--   2 = Progressive (YTD only)
--   3 = Combined
--   4 = All
--   5 = P3F View: Actual, P3F, v P3F, Budget, v Budget, then YTD set
--   6 = P6F View: same layout for P6F
--   7 = P9F View: same layout for P9F
-- ============================================================
DATATABLE(
    "View ID",   INTEGER,
    "View Name", STRING,
    "View Sort", INTEGER,
    {
        {1, "Current Month", 1},
        {2, "Progressive",   2},
        {3, "Combined",      3},
        {4, "All",           4},
        {5, "P3F View",      5},
        {6, "P6F View",      6},
        {7, "P9F View",      7}
    }
)
