        ([Year Flag] = "P6F" and [Period] <= 6) or
        ([Year Flag] = "P9F" and [Period] <= 9))),

#"TY Base" = Table.SelectRows(#"Buffered", each [Year Flag] = "TY" and [Period] <= 9),

        Table.TransformColumns(Table.SelectRows(#"TY Base", each [Period] <= 3),

    else Table.FirstN(#"TY Base", 0),

#"P6F Actuals" =
    if #"P6F Loaded" then
        Table.TransformColumns(Table.SelectRows(#"TY Base", each [Period] <= 6),
            {{"Year Flag", each "P6F", type text}, {"Version", each "Actuals in P6F", type text}})
    else Table.FirstN(#"TY Base", 0),

#"P9F Actuals" =
    if #"P9F Loaded" then
        Table.TransformColumns(#"TY Base",
            {{"Year Flag", each "P9F", type text}, {"Version", each "Actuals in P9F", type text}})
    else Table.FirstN(#"TY Base", 0),
