EVALUATE VAR _CCs = SELECTCOLUMNS('Marketing', "CC", 'Marketing'[GL]) VAR _Mapped = SELECTCOLUMNS( FILTER( SUMMARIZE( CALCULATETABLE( 'FS SPL (fact)', 'FS SPL (fact)'[SPL Line source] = "IO", TREATAS(_CCs, 'FS SPL (fact)'[Cost Centre]) ), 'FS SPL (fact)'[IO Name] ), NOT ISBLANK('FS SPL (fact)'[IO Name]) ), "IO", 'FS SPL (fact)'[IO Name] ) VAR _Others = FILTER( CALCULATETABLE('FS SPL (fact)', TREATAS(_CCs, 'FS SPL (fact)'[Cost Centre])), ISBLANK('FS SPL (fact)'[IO Name]) || NOT ('FS SPL (fact)'[IO Name] IN _Mapped) ) RETURN SUMMARIZE( _Others, 'FS SPL (fact)'[Cost Centre], 'FS SPL (fact)'[GL], 'FS SPL (fact)'[Doc type], 'FS SPL (fact)'[IO Name], 'FS SPL (fact)'[Year Flag], "Amount", SUM('FS SPL (fact)'[Amount]) ) ORDER BY 'FS SPL (fact)'[Cost Centre], 'FS SPL (fact)'[GL]
